% CHAPTERS AND SECTIONS

% No numbering by default.

\setcounter{secnumdepth}{0}

\makeatletter

% Chapter headings start a new page, as full width, broken-out, flush left,
% ragged right, \Large, bold, and uppercase. Chapters are never numbered.
% Chapters are not handled by \@startsection, so we have to redefine them
% manually.

\newcommand{\chapter@star}[1]{%
    \stepcounter{chapter}%
    \clearpage%
    \twocolumn[%
        \vspace*{1.5ex}%
        \normalfont{\Large\bfseries\raggedright\MakeUppercase{#1}}\par
        \vspace{2.0ex}%
        \phantomsection
    ]%
}
\newcommand{\chapter@nostar}[1]{%
    \chapter@star{#1}%
    \addcontentsline{toc}{chapter}{\MakeUppercase{#1}}%
}
\renewcommand{\chapter}{\@ifstar{\chapter@star}{\chapter@nostar}}

% Section headings are broken-out, flush left, ragged right, \normalsize, bold,
% and uppercase.

\newcommand{\section@handler}{%
    \@startsection{section}{1}%
    {0pt}%
    {2.0ex plus 0.6ex minus 0.2ex}%
    {1.0ex plus 0.3ex minus 0.1ex}%
    {\normalfont\normalsize\bfseries\raggedright}%
}
\newcommand{\section@star}[1]{%
    \section@handler*{\MakeUppercase{#1}}%
}
\newcommand{\section@nostar}[1]{%
    \section@handler{\MakeUppercase{#1}}%
}
\renewcommand{\section}{\@ifstar{\section@star}{\section@nostar}}

% Subsections headings are run-in, indented, normalsize, bold, with a trailing
% colon.

\renewcommand{\subsection}{%
    \@startsection{subsection}{2}%
    {\parindent}%
    {0pt}%
    {-\interwordspace}%
    {\normalfont\normalsize\bfseries\@addcolon}%
}
\newcommand{\@addcolon}[1]{#1:}
\newlength{\interwordspace}
\setlength{\interwordspace}{
    \fontdimen2\font plus \fontdimen3\font minus \fontdimen4\font
}

% Paragraph headings are run-in, indented, normalsize, italic, with a trailing
% colon.

\renewcommand\paragraph{%
    \@startsection{paragraph}{4}%
    {\parindent}%
    {0pt}%
    {-\interwordspace}%
    {\normalfont\normalsize\itshape\@addcolon}%
}

\makeatother

% columns).

% The multicols package can correctly set the TOC in a single column, but it
% comes with severe restrictions on the use of floats.

% This implementation has a bug: text after the TOC is set to half the width of
% a column. This can be fixed by use of \twocolumn again.

\makeatletter

\newlength{\saved@columnwidth}
\newlength{\saved@linewidth}

    \setlength{\saved@textwidth}{\textwidth}
    \setlength{\saved@columnwidth}{\columnwidth}

    \setlength{\textwidth}{\linewidth}
    %
    \let\saved@clearpage\clearpage
    \@ifundefined{chapter}{}{\let\saved@chapter\chapter}

% ALIGNMENT AND SPACING

\makeatletter

% The \wbox command puts its second argument in a box equal in width to its
% first argument. The alignment of the contents of the box can be specified by
% an optional argument and defaults to r.

\newlength{\@wboxwidth}
\newcommand{\wbox}[3][r]{%
    \settowidth{\@wboxwidth}{#2}%
    \makebox[\@wboxwidth][#1]{#3}%
}

% The \fitmaxwidthbox command puts its second argument whose width is the
% smaller of the natural width of the second argument or the length specified by
% the first argument. The alignment of the contents of the box is r.

\newlength{\@fitmaxwidthboxlength}
\newcommand{\fitmaxwidthbox}[2]{%
    \settowidth{\@fitmaxwidthboxlength}{#2}%
    \resizebox{\minof{\@fitmaxwidthboxlength}{#1}}{!}{#2}%
}

% The fitwidth environment puts its content in a box whose width is the first
% argument.

\newsavebox{\@fitwidthbox}
\newenvironment{fitwidth}[1]{%
    \def\@fitwidth{#1}%
    \begin{lrbox}{\@fitwidthbox}%
    }{%
    \end{lrbox}%
    \resizebox{\@fitwidth}{!}{\usebox{\@fitwidthbox}}%
}

% The fitwidth environment puts its content in a box whose height is the first
% argument.

\newsavebox{\@fitheightbox}
\newenvironment{fitheight}[1]{%
    \def\@fitheight{#1}%
    \begin{lrbox}{\@fitheightbox}%
    }{%
    \end{lrbox}%
    \resizebox{!}{\@fitheight}{\usebox{\@fitheightbox}}%
}

\makeatother
